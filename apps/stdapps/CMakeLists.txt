# Each stdapp is a directory of *.c files (ardubot create-app/install both
# only need to add the directory - no source list to edit here).
set(ARDUBOT_ALL_STDAPPS
    launcher
    counter
    info
    stopwatch
    pong
    settings
    fileman
    shell
    demo
)

# A device profile can restrict what actually gets compiled in, via
# -DARDUBOT_ENABLED_APPS="counter;info;pong" (semicolon-separated, CMake
# list syntax). This is how "based on the device, only compile what fits"
# is implemented: scripts/device_config.py resolve_apps() computes this
# list from device_config.yaml (explicit `apps:` override, or auto-fit
# against each app's package.json "min_display"), and the Makefile passes
# it through CMAKE_OPTS for ARCH != sim. Unset (the sim default) means
# "compile everything."
if(NOT DEFINED ARDUBOT_ENABLED_APPS)
    set(ARDUBOT_ENABLED_APPS ${ARDUBOT_ALL_STDAPPS})
endif()

# launcher is the home app - always required regardless of what a device
# profile asked for, so a bad/empty `apps:` list can't produce an
# unbootable image.
if(NOT "launcher" IN_LIST ARDUBOT_ENABLED_APPS)
    list(APPEND ARDUBOT_ENABLED_APPS "launcher")
endif()

set(STDAPP_SOURCES "")
set(STDAPP_INCLUDE_DIRS "")
foreach(app ${ARDUBOT_ENABLED_APPS})
    if(NOT app IN_LIST ARDUBOT_ALL_STDAPPS)
        message(WARNING "ardubot: ARDUBOT_ENABLED_APPS names unknown stdapp '${app}', skipping")
        continue()
    endif()
    file(GLOB app_srcs "${CMAKE_CURRENT_SOURCE_DIR}/${app}/*.c")
    list(APPEND STDAPP_SOURCES ${app_srcs})
    list(APPEND STDAPP_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/${app}")
endforeach()

add_library(stdapps STATIC ${STDAPP_SOURCES})

target_include_directories(stdapps PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${STDAPP_INCLUDE_DIRS}
    ${CMAKE_SOURCE_DIR}/apps
    ${CMAKE_SOURCE_DIR}/kernel
    ${CMAKE_SOURCE_DIR}/drivers
    ${CMAKE_SOURCE_DIR}/fs
    ${CMAKE_SOURCE_DIR}/hal/include
    ${CMAKE_SOURCE_DIR}/sim/models
    ${CMAKE_SOURCE_DIR}/sim
)

target_link_libraries(stdapps PUBLIC apps)

message(STATUS "ardubot: stdapps enabled = ${ARDUBOT_ENABLED_APPS}")

# Anything that references an app's manifest (sim_main.c today, a future
# board main.c tomorrow) needs to know at compile time which ones actually
# got built - otherwise an excluded app's extern app_manifest_t* is a link
# error. One #define per enabled app; absent means "not compiled in."
set(ENABLED_APPS_HEADER "${CMAKE_BINARY_DIR}/generated/ardubot_enabled_apps.h")
file(WRITE "${ENABLED_APPS_HEADER}"
    "#pragma once\n/* Generated by apps/stdapps/CMakeLists.txt from ARDUBOT_ENABLED_APPS - do not edit. */\n")
foreach(app ${ARDUBOT_ENABLED_APPS})
    if(app IN_LIST ARDUBOT_ALL_STDAPPS)
        string(TOUPPER "${app}" app_upper)
        file(APPEND "${ENABLED_APPS_HEADER}" "#define ARDUBOT_APP_${app_upper}_ENABLED 1\n")
    endif()
endforeach()
